# Propriedades principais
# Box da pagina 
margin: espaço externo
padding: espaço interno
border: borda
box-sizing: controla cálculo de tamanho

# Cores/fundo


color: cor do texto
background: fundo geral
background-color: cor de fundo
background-image: imagem de fundo
background-size: tamanho da imagem de fundo
background-position: posição da imagem
background-repeat: repetição da imagem

# Texto

font-size: tamanho da fonte
font-weight: peso da fonte
font-family: tipo de fonte
text-align: alinhamento do texto
line-height: altura da linha
text-decoration: decoração (underline, etc)
text-transform: maiúscula/minúscula

# Tamanho
margin: espaço externo
margin-top: margem superior
margin-bottom: margem inferior
margin-left: margem esquerda
margin-right: margem direita

padding: espaço interno
padding-top: interno superior
padding-bottom: interno inferior
padding-left: interno esquerda
padding-right: interno direita

# Layout da pagina

display: flex
flex-direction: direção (row/column)
justify-content: alinhamento horizontal
align-items: alinhamento vertical
gap: espaço entre elementos
flex-wrap: quebra de linha
flex-direction: linha ou coluna

# layout grid

display: grid
grid-template-columns: colunas
grid-template-rows: linhas
gap: espaçamento

# Interação com os itens

cursor: tipo do cursor
hover: efeito ao passar o mouse
transition: animação suave

# Classes uteis padrão

.container: centraliza conteúdo 
.card: componente reutilizável
.btn: botão padrão
.hidden: esconder elemento
.flex: ativa flexbox
.grid: ativa grid
